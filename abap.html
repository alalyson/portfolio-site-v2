01

FORM save_deposit .
  DATA : ls_save  TYPE zte2fi0012,
         lt_save  LIKE TABLE OF ls_save,
         lv_tabix TYPE sy-index,

  CLEAR : lv_tabix, gt_docs.

    ls_save-status = 'N'.

      ls_save-erdat = sy-datum.
      ls_save-erzet = sy-uzeit.

    ELSE.
      ls_save-aedat = sy-datum.

      ls_save-aenam = sy-uname.
    ENDIF.
    CLEAR gs_current_doc.
    gs_current_doc-header-bukrs = '1000'.
    gs_current_doc-header-gjahr = sy-datum(4).
    gs_current_doc-header-blart = 'DZ'. " 입금
    gs_current_doc-header-bldat = gs_deposit-ban_dat.
    gs_current_doc-header-budat = gs_deposit-ban_dat.
    gs_current_doc-header-waers = gs_deposit-waers.
    gs_current_doc-header-xblnr = ls_save-seq_no. " 참조번호에 일련번호 매핑
    ls_item_tmp = VALUE #( bschl = '40'
                           shkzg = 'S'
                           hkont = '0010000002'
                           wrbtr = gs_deposit-d_amt
                           sgtxt = |[고객 입금] { gs_deposit-contact }| ).
    APPEND ls_item_tmp TO gs_current_doc-items.
    ls_item_tmp = VALUE #( bschl = '50'
                           shkzg = 'H'
                           hkont = '0020000004' " 임시 지정한 가수금 계정
                           wrbtr = gs_deposit-d_amt
                           sgtxt = |[입금 가계정] { gs_deposit-contact }| ).
    APPEND ls_item_tmp TO gs_current_doc-items.
    APPEND gs_current_doc TO gt_docs.
    APPEND ls_save TO lt_save.
    CLEAR  ls_save.
  ENDLOOP.
  IF gt_docs IS NOT INITIAL.
    CALL METHOD zcl_e2_fi_posting=>create_fi_multi_document
      EXPORTING
        iv_wrttp      = '04' " 실제 데이터
      IMPORTING
        ev_last_belnr = gv_last_belnr
        et_fi_head    = gt_fi_head
        et_fi_item    = gt_fi_item
      CHANGING
        ct_docs       = gt_docs.
    LOOP AT lt_save ASSIGNING FIELD-SYMBOL(<fs_save>).
      READ TABLE gt_fi_head INTO DATA(ls_head_map) WITH KEY xblnr = <fs_save>-seq_no.
      IF sy-subrc = 0.
        <fs_save>-belnr = ls_head_map-belnr.
      ENDIF.
    ENDLOOP.
  ENDIF.
  IF lt_save IS NOT INITIAL.
    MODIFY zte2fi0012 FROM TABLE lt_save.
    IF sy-dbcnt > 0.
      COMMIT WORK AND WAIT.
      MESSAGE s002.
    ENDIF.
  ENDIF.
ENDFORM.

FORM calculate_payroll.
  DATA: lv_monthly TYPE p DECIMALS 2.
  REFRESH: gt_dept_sum, gt_alv.
  SORT gt_employee BY kostl.
  LOOP AT gt_employee INTO DATA(ls_emp).
    lv_monthly = ls_emp-ansal / 1200.
    READ TABLE gt_dept_sum ASSIGNING FIELD-SYMBOL(<fs_sum>) WITH KEY kostl = ls_emp-kostl.
    IF sy-subrc = 0.
      <fs_sum>-total_amt = <fs_sum>-total_amt + lv_monthly.

      <fs_sum>-kostl     = ls_emp-kostl.
      <fs_sum>-waers     = ls_emp-waers.
      <fs_sum>-total_amt = lv_monthly.
      <fs_sum>-emp_cnt   = 1.
    ENDIF.
  ENDLOOP.
  LOOP AT gt_dept_sum INTO DATA(ls_sum).
    gs_alv-status    = icon_led_green.
    gs_alv-kostl     = ls_sum-kostl.
    gs_alv-emp_cnt   = ls_sum-emp_cnt.

    ENDIF.
    APPEND gs_alv TO gt_alv.
  ENDLOOP.

FORM pull_pp_data .
  DATA: lv_budat_from TYPE zte2pp0011-iedd,
        lv_budat_to   TYPE zte2pp0011-iedd.
  DATA: lt_sval TYPE TABLE OF sval,
        ls_sval TYPE sval.
  DATA: lv_returncode TYPE c.

  ls_sval-fieldname = 'GJAHR'.

  ls_sval-field_obl = 'X'.
  APPEND ls_sval TO lt_sval.

    EXPORTING

    IMPORTING

  DATA(lv_month) = CONV i( ls_sval-value ).

  lv_budat_from = |{ gs_skftr-gjahr }{ lv_month WIDTH = 2 ALIGN = RIGHT PAD = '0' }01|.

    EXCEPTIONS

수익성 계산에는 자재 유형 확인과 CO 표준원가 또는 SD·MM 배치 단가 조회가 필요합니다. 일반 동기 함수 호출은 이 계산이 끝나야 전표 생성 로직으로 돌아옵니다. tRFC는 호출 시 실행을 등록하고 COMMIT WORK에서 시작하므로, 전표 생성 중 계산 결과를 즉시 기다리는 구조를 피하고 수익성 계산을 후속 처리로 분리할 수 있습니다.

적용 효과 · 커밋 전 ROLLBACK WORK가 수행되면 등록된 호출도 취소되어, 취소한 전표의 수익성 계산이 실행되는 것을 방지합니다. 실행 시점을 전표 확정에 맞추되, 전표 저장 성공과 후속 계산 성공은 별도로 확인해야 합니다.

수익성 데이터 흐름

다중 전표 생성 → RFC 수익성 계산 → 분석 데이터 저장 → CDS 가공·집계 → OData → UI5

함수 호출과 계산 로직은 소스 발췌로 확인할 수 있습니다. 표준원가 산정 및 SD·MM 데이터는 팀 프로그램과의 연계 대상입니다.

METHOD period_check.
  DATA: lv_status TYPE zte2co0009-co_status.
  CLEAR rv_check.
  SELECT SINGLE co_status
    FROM zte2co0009
   WHERE bukrs    = @iv_bukrs
     AND gjahr    = @iv_gjahr
     AND monat    = @iv_monat
     AND co_module = @iv_module
    INTO @lv_status.
  IF sy-subrc = 0 AND lv_status = 'C'.
    rv_check = 'C'. " 진행 불가(Closed) 상태 반환
    MESSAGE |{ iv_gjahr }년 { iv_monat }월의 { iv_module } 모듈 결산이 마감되어 전표/수불을 생성할 수 없습니다.|
    TYPE 'S' DISPLAY LIKE 'E'.
    EXIT.
  ELSE.
    rv_check = 'O'.
  ENDIF.
ENDMETHOD.
